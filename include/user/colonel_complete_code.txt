// SPDX-License-Identifier: GPL-2.0
/*
 * sprof_eud.c
 *
 * Qualcomm EUD (Enterprise User Data)
 * interface for reading actual hardware
 * values from Snapdragon 6450
 *
 * This file reads the REAL hardware values
 * from your device then those values get
 * used to build the SPOOFED content that
 * makes binaries think they are on a
 * different platform
 *
 * The flow is:
 *   EUD reads real Snapdragon 6450 values
 *   Profile uses those values as a base
 *   Content generation creates spoofed
 *   /proc/* files that LOOK like the
 *   target platform but use real hardware
 *   characteristics as the foundation
 *
 * This eliminates wasted cycles because:
 *   1. Real hardware values are used
 *      (no guessing no static values)
 *   2. One read at profile creation time
 *      (not on every /proc read)
 *   3. No failed installations from
 *      wrong hardware detection
 */

#include <linux/module.h>
#include <linux/kernel.h>
#include <linux/init.h>
#include <linux/slab.h>
#include <linux/string.h>
#include <linux/platform_device.h>
#include <linux/io.h>
#include <linux/of.h>
#include <linux/of_device.h>
#include <linux/cpufreq.h>
#include <linux/cpu.h>
#include <linux/mm.h>
#include <linux/swap.h>
#include <linux/utsname.h>
#include <linux/netdevice.h>
#include <linux/etherdevice.h>

#include "sprof_core.h"

/* =========================================================
 * Qualcomm EUD register offsets
 * These are memory-mapped registers that
 * provide hardware information on Qualcomm
 * Snapdragon platforms
 *
 * On Snapdragon 6450 these are located
 * in the QFPROM (Qualcomm Fuse Proximity)
 * address space
 * ========================================================= */

/* QFPROM base address for Snapdragon 6450 */
#define QCOM_QFPROM_BASE       0x00780000
#define QCOM_QFPROM_SIZE       0x10000

/* Multiplier register for CPU speed */
#define QFPROM_OEM_ID_OFFSET   0x000000E0
#define QFPROM_PTE_OFFSET      0x00000100

/* SOC version register */
#define QFPROM_SOC_VER_OFFSET  0x000000E8

/* CPU Part Number offsets in QFPROM */
#define QFPROM_CPU_PART_OFFSET 0x000000F0
#define QFPROM_CPU_VARIANT_OFFSET 0x000000F4

/* =========================================================
 * Snapdragon 6450 specific definitions
 * These values are specific to the SM6450
 * chipset also known as "Taro" or "Lanai"
 * ========================================================= */

/* Snapdragon 6450 part number */
#define SM6450_PART_NUM        0x00E0
#define SM6450_VARIANT         0x0001
#define SM6450_OEM_ID          0x0051
#define SM6450_SOC_VER         0x60050000

/* SM6450 CPU topology */
/* 4x Cortex-A78 @ 2.4 GHz */
/* 4x Cortex-A55 @ 1.8 GHz */
#define SM6450_MAX_FREQ_KHZ    2400000
#define SM6450_MIN_FREQ_KHZ    1800000
#define SM6450_BIG_CORES       4
#define SM6450_LITTLE_CORES    4
#define SM6450_TOTAL_CORES     8

/* =========================================================
 * Internal EUD state
 * ========================================================= */

struct qcom_eud_state {
    void __iomem    *qfprom_base;
    int             qfprom_mapped;
    int             initialized;
    spinlock_t      lock;

    /* Cached hardware values read from EUD */
    int             cpu_family_real;
    int             cpu_model_real;
    int             cpu_variant_real;
    int             cpu_part_real;
    int             oem_id_real;
    int             soc_ver_real;

    /* Cached from cpufreq */
    int             max_freq_khz;
    int             min_freq_khz;

    /* Cached from platform */
    int             total_cores;
    int             big_cores;
    int             little_cores;
};

static struct qcom_eud_state eud_state;

/* =========================================================
 * EUD Low-level read functions
 * Read raw values from Qualcomm registers
 * ========================================================= */

static inline u32 qfprom_read(u32 offset)
{
    if (!eud_state.qfprom_mapped)
        return 0;

    return readl(eud_state.qfprom_base + offset);
}

/* =========================================================
 * EUD CPU reading
 * Read actual CPU information from the
 * Snapdragon 6450
 * ========================================================= */

int sprof_qcom_eud_read_cpu(
    struct sprof_hw_profile *hw)
{
    u32 reg_val;
    int online_cpus;
    int cpu;

    if (!hw)
        return -EINVAL;

    if (!eud_state.initialized) {
        pr_warn("sprof_eud: not initialized\n");
        return -ENODEV;
    }

    /*
     * Read QFPROM registers for CPU identification
     * These give us the REAL silicon values
     * that prove this is a Snapdragon 6450
     */
    reg_val = qfprom_read(QFPROM_CPU_PART_OFFSET);
    eud_state.cpu_part_real = reg_val & 0xFFFF;

    reg_val = qfprom_read(QFPROM_CPU_VARIANT_OFFSET);
    eud_state.cpu_variant_real = reg_val & 0xFFFF;

    reg_val = qfprom_read(QFPROM_SOC_VER_OFFSET);
    eud_state.soc_ver_real = reg_val;

    reg_val = qfprom_read(QFPROM_OEM_ID_OFFSET);
    eud_state.oem_id_real = reg_val & 0xFFFF;

    pr_info("sprof_eud: QFPROM read complete\n");
    pr_info("sprof_eud: cpu_part=0x%04x\n",
        eud_state.cpu_part_real);
    pr_info("sprof_eud: cpu_variant=0x%04x\n",
        eud_state.cpu_variant_real);
    pr_info("sprof_eud: soc_ver=0x%08x\n",
        eud_state.soc_ver_real);
    pr_info("sprof_eud: oem_id=0x%04x\n",
        eud_state.oem_id_real);

    /*
     * Read CPU frequency information
     * from cpufreq subsystem
     * This gives us the REAL clock speeds
     */
    eud_state.max_freq_khz = cpufreq_get(0);
    if (eud_state.max_freq_khz <= 0)
        eud_state.max_freq_khz = SM6450_MAX_FREQ_KHZ;

    eud_state.min_freq_khz = SM6450_MIN_FREQ_KHZ;

    pr_info("sprof_eud: max_freq=%d kHz\n",
        eud_state.max_freq_khz);

    /*
     * Count online CPUs to determine
     * actual active core count
     */
    online_cpus = 0;
    for_each_online_cpu(cpu) {
        online_cpus++;
    }
    eud_state.total_cores = online_cpus;

    if (eud_state.total_cores > SM6450_BIG_CORES) {
        eud_state.big
